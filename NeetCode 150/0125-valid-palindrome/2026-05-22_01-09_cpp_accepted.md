## Solution

```cpp
class Solution {
public:
    bool isAlphaNumeric(char ch){
        return  ('A' <= ch && ch <= 'Z') || 
                ('a' <= ch && ch <= 'z') || 
                ('0' <= ch && ch <= '9');
    }

    bool isPalindrome(string s) {
        int i = 0, j = s.size()-1;

        while(i<j){
            // Move i unless it get's a valid char
            while(i<j && !isAlphaNumeric(s[i]))    i++;
            while(i<j && !isAlphaNumeric(s[j]))    j--;

            if(tolower(s[i]) != tolower(s[j]))    return false;
            i++;
            j--;
        }

        return true;
    }
};
```
