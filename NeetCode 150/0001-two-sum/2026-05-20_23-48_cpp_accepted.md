## Solution

```cpp
class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        // Using running hashmap to effectively find the requiredNum which sums up to target for every num
        unordered_map<int, int> hash;
        for(int i=0; i<nums.size(); i++){
            int requiredNum = target - nums[i];
            if(hash.find(requiredNum) != hash.end()){
                return {hash[requiredNum], i};
            }
            hash[nums[i]] = i;
        }
        return {};
    }
};
```
