## Solution

```cpp
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        // Using 2 pointer approach
        // left pointer for minimum price => buy day
        // right pointer for maximum price => sell day

        int n = prices.size();
        int left = 0, right = 1;
        int maxProfit = 0;

        while(right < n){
            if(prices[right] > prices[left]){
                // Possible sell day
                int profit = prices[right] - prices[left];
                maxProfit = max(maxProfit, profit);
            }else{
                // Buy day
                left = right;
            }

            right++;
        }

        return maxProfit;
    }
};
```
