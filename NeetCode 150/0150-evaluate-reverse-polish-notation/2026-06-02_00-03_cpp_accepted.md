## Solution

```cpp
class Solution {
public:
    int evalRPN(vector<string>& tokens) {
        // Use stack approach
        // Keep pushing number on to the stack. Once we see an operator, 
        // we pop recent 2 operands, perform the operation and push it back on to the stack, proceed

        stack<long long> st;

        for (string& token : tokens) {

            // If token is an operator
            if (token == "+" || token == "-" ||
                token == "*" || token == "/") {

                long long operand2 = st.top(); st.pop();
                long long operand1 = st.top(); st.pop();

                long long result = 0;

                if (token == "+")
                    result = operand1 + operand2;
                else if (token == "-")
                    result = operand1 - operand2;
                else if (token == "*")
                    result = operand1 * operand2;
                else
                    result = operand1 / operand2;

                st.push(result);
            }
            else {
                // Token is a number
                st.push(stoll(token));
            }
        }

        return (int)st.top();
    }
};
```
