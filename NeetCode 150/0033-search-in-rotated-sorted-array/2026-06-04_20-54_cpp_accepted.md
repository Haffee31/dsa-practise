## Solution

```cpp
class Solution {
public:
    int search(vector<int>& nums, int target) {
        // Using Binary search to eliminate the half based on correctly sorted half

        int n = nums.size();
        int low = 0, high = n-1;

        while(low <= high){
            int mid = low + (high - low) / 2;

            if(nums[mid] == target)
                return mid;

            // Left side is sorted
            if(nums[low] <= nums[mid]){
                // Check if target lies in left side
                if(nums[low] <= target && target <= nums[mid]){
                    high = mid - 1;
                }else{
                    low = mid + 1;
                }
            } // Right side is sorted
            else{
                // Check if target lies in right side
                if(nums[mid] <= target && target <= nums[high]){
                    low = mid + 1;
                }else{
                    high = mid - 1;
                }
            }
        }

        return -1;
    }
};
```
