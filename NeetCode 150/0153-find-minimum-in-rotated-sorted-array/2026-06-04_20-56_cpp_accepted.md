## Solution

```cpp
class Solution {
public:
    int findMin(vector<int>& nums) {
        // Using Binary search to eliminate the half based on correctly sorted half

        int low = 0, high = nums.size() - 1;
        int minimum = nums[0];
        
        while(low <= high){
            int mid = low + (high - low) / 2;

            // Optimization trick
            if(nums[low] <= nums[high]){
                minimum = min(minimum, nums[low]);
                break;
            }

            // Correctly sorted. Current minimum can be our minimum and move left part to find if we can find a better minimum
            if(nums[low] <= nums[mid]){
                minimum = min(minimum, nums[low]);
                low = mid + 1;
            }else{
                minimum = min(minimum, nums[mid]);
                high = mid - 1;
            }
        }

        return minimum;
    }
};
```
