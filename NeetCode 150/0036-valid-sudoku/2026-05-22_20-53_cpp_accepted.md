## Solution

```cpp
class Solution {
public:
    bool isValidSudoku(vector<vector<char>>& board) {
        int m = 9;
        int n = 9;

        // Vector of hashsets for faster lookups in rows, cols and squares (mathematically drawn)
        vector<unordered_set<char>> rows(m);
        vector<unordered_set<char>> cols(n);
        vector<unordered_set<char>> squares(n);

        for(int i=0; i<m; i++){
            for(int j=0; j<n; j++){
                if(board[i][j] == '.')
                    continue;

                char digit = board[i][j];
                int squareNumber = (3 * (i / 3)) + (j / 3);
                if(rows[i].count(digit) > 0 || 
                    cols[j].count(digit) > 0 || 
                    squares[squareNumber].count(digit) > 0){
                        return false;
                }

                rows[i].insert(digit);
                cols[j].insert(digit);
                squares[squareNumber].insert(digit);
            }
        }

        return true;
    }
};
```
