## Solution

```cpp
class Solution {
public:
    int maxArea(vector<int>& height) {
        // Two Pointers Approach
        int left = 0, right = height.size()-1;

        // water = Area of Container => (height x width)

        // Main point to remember: Whether we move left pointer or right pointer, we are decreasing the width
        // So, to get more area, greedily we want to get more height since Area = height x width
        // Therefore, to get more height, we keep the max height bar with us and move the min height bar expecting to get a bigger bar

        // height = min(left, right)
        // width = right - left

        int maxWater = 0;
        while(left < right){
            // calculating current water holding
            int h = min(height[left], height[right]);
            int w = right - left;
            int currentWater =  h * w;
            maxWater = max(maxWater, currentWater);

            // if left bar is smaller will move to right expecting to get a bigger block. i.e retain right block
            if(height[left] <= height[right])   left++;
            // vice versa
            else    right--;

        }

        return maxWater;
    }
};
```
