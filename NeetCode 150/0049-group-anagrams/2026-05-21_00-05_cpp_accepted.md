## Solution

```cpp
class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        // Normalize data into a canonical form = Character Frequency Hashing

        // For Every string, we build a count array (No sorting since 26 characters only) which is efficient
        // Later this count array is converted/serialized into a string with CharCount#CharCount#...
        
        unordered_map<string, vector<string>> mp;
        for(string& str: strs){
            int count[26] = {0};

            // Freq of each char of the string
            for(char& ch: str){
                count[ch - 'a']++;
            }

            // Serialize the count array into a string
            string key = "#";
            for(int i=0; i<26; i++){
                key += (char)(i + 'a') + to_string(count[i]) + '#';
            }

            mp[key].push_back(str);  // Key is serialized string and the value is the original string
        }

        // Collect all groups
        vector<vector<string>> groups;
        for(auto &[_, stringGroup]: mp){
            groups.push_back(stringGroup);
        }

        return groups;
    }
};
```
