## Solution

```cpp
class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        // This problem is nothing but Next Greater Element (NGE)
        int n = temperatures.size();
        stack<int> st;
        vector<int> result(n, 0);

        for(int i=n-1; i>=0; i--){

            while(!st.empty() && temperatures[st.top()] <= temperatures[i]){
                st.pop();
            }

            if(!st.empty()){
                result[i] = st.top() - i;   // Number of days to wait
            }

            st.push(i);
        }

        return result;
    }
};
```
