## Solution

```cpp
class Solution {
public:
    double findMedianSortedArrays(vector<int>& nums1, vector<int>& nums2) {

        // Always binary search on the smaller array
        if (nums1.size() > nums2.size()) {
            return findMedianSortedArrays(nums2, nums1);
        }

        int m = nums1.size();
        int n = nums2.size();
        int total = m + n;

        int left = 0;
        int right = m;

        while (left <= right) {

            // Partition positions
            int partition1 = left + (right - left) / 2;
            int partition2 = (total + 1) / 2 - partition1;

            // Elements around the partitions
            int left1  = (partition1 == 0) ? INT_MIN : nums1[partition1 - 1];
            int right1 = (partition1 == m) ? INT_MAX : nums1[partition1];

            int left2  = (partition2 == 0) ? INT_MIN : nums2[partition2 - 1];
            int right2 = (partition2 == n) ? INT_MAX : nums2[partition2];

            // Valid partition found
            if (left1 <= right2 && left2 <= right1) {

                int maxLeft = max(left1, left2);
                int minRight = min(right1, right2);

                if (total % 2 == 1) {
                    return maxLeft;
                }

                return (maxLeft + minRight) / 2.0;
            }

            // Move partition1 to the left
            if (left1 > right2) {
                right = partition1 - 1;
            }
            // Move partition1 to the right
            else {
                left = partition1 + 1;
            }
        }

        return 0.0; // Unreachable for valid input
    }
};
```
