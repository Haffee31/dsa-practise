## Solution

```cpp
class Solution {
public:
    bool isValid(string s) {
        // Use stack to track the order as well as correct open bracket for the closed one.
        stack<char> st;

        for (int i = 0; i < s.length(); i++) {
            if (s[i] == '(' || s[i] == '{' || s[i] == '[') {
                st.push(s[i]);
            } else {
                if (st.empty()) // Don't forget this case
                    return false;

                char topChar = st.top();
                st.pop();

                if ((s[i] == ')' && topChar != '(') ||
                    (s[i] == '}' && topChar != '{') ||
                    (s[i] == ']' && topChar != '['))
                    return false;
            }
        }

        return st.empty();  // Final case
    }
};
```
