## Solution

```cpp
class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        // Use hash set and iterate on hashset for faster lookup, so see if the next or prev num is present and find longest sequence length

        unordered_set<int> us(nums.begin(), nums.end());
        int maxSequenceLen = 0;

        for(int num: us){   // Iterate on set instead of nums VVV IMP
            if(us.find(num - 1) != us.end())
                continue;   // Will find the longest sequence when we reach the previous num in the array

            int sequenceLen = 0;
            while(us.find(num) != us.end()){
                sequenceLen++;

                num++;
            }

            maxSequenceLen = max(maxSequenceLen, sequenceLen);
        }

        return maxSequenceLen;
    }
};
```
