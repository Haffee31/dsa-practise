## Solution

```cpp
class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        // Using sliding window and hashmap for char => index 

        int n = s.size();
        int right = 0, left = 0;

        unordered_map<char, int> hashMap;
        int maxLen = 0;

        while(right < n){
            // If already present and it is inside the considered window => duplicate
            if(hashMap.find(s[right]) != hashMap.end() && hashMap[s[right]] >= left){
                left = hashMap[s[right]] + 1;
            }

            hashMap[s[right]] = right;

            maxLen = max(maxLen, right - left + 1);
            right++;
        }

        return maxLen;
    }
};
```
