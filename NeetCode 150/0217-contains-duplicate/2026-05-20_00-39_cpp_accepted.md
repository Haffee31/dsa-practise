## Solution

```cpp
class Solution {
public:
    bool containsDuplicate(vector<int>& nums) {
        // Brute force => For every element nums[i],
        // find the occurences in the entire array, if it is greater than 1 =>
        // contains duplicate O(n^2) solution

        // Optimal Solution
        // Use hashSet to put the numbers into the set. If a number is already
        // present, the current one is a duplicate TC: O(n) SC: O(n)

        unordered_set<int> seen;
        for (int num : nums) {
            if (seen.find(num) != seen.end()) {
                return true; // Duplicate found
            }

            seen.insert(num);
        }

        return false;
    }
};
```
