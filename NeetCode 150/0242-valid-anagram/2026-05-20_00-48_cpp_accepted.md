## Solution

```cpp
class Solution {
public:
    bool isAnagram(string s, string t) {
        // Optimal solution:
        // Use hash tables to check the frequency of characters
        // TC: O(m + n)
        // SC: O(1) => since only 26 characters. If not, we have to use hashmap<char, int>

        // s is anagaram of t (frequency of characters of s == frequency of characters of t)
        
        // 1. check if lengths  are same
        if(s.length() != t.length())    return false;
        
        // using frequency array for faster access
        int freqArray[26] = {0};
        
        // increase count for s[i] and decrease count for t[i]
        for(int i=0; i<s.length(); i++){
            freqArray[s[i]-'a']++;
            freqArray[t[i]-'a']--;
        }
        
        // even if single character count is non zero then return false
        // else return true
        for(int i=0; i<26; i++)
            if(freqArray[i] != 0)   return false;
        
        return true;
    }
};
```
