## Solution

```cpp
class Solution {
public:
    int carFleet(int target, vector<int>& position, vector<int>& speed) {

        int n = position.size();

        // Store each car as {position, speed}
        vector<pair<int, int>> cars;

        for (int i = 0; i < n; i++) {
            cars.push_back({position[i], speed[i]});
        }

        // Sort cars by position so that we can process
        // from the car closest to the target to the farthest.
        sort(cars.begin(), cars.end());

        int fleets = 0;

        // Arrival time of the last fleet formed.
        double fleetTime = 0;

        // Process cars from right to left.
        for (int i = n - 1; i >= 0; i--) {

            // Time needed for current car to reach target.
            double currTime =
                (double)(target - cars[i].first) / cars[i].second;

            // If current car arrives later than the fleet ahead,
            // it cannot catch that fleet, so it forms a new fleet.
            if (currTime > fleetTime) {
                fleets++;
                fleetTime = currTime;
            }

            // Otherwise:
            // currTime <= fleetTime
            // Current car catches the fleet ahead and becomes
            // part of that fleet. No new fleet is formed.
        }

        return fleets;
    }
};
```
