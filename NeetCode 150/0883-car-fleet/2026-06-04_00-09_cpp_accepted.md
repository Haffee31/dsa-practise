## Solution

```cpp
class Solution {
public:
    int carFleet(int target, vector<int>& position, vector<int>& speed) {

        int n = position.size();

        // Store each car as {position, speed}
        vector<pair<int, int>> cars;

        for (int i = 0; i < n; i++) {
            cars.push_back({position[i], speed[i]});
        }

        // Sort cars by position so that we can process
        // from the car closest to target to the farthest.
        sort(cars.begin(), cars.end());

        // Stack stores arrival times of fleets.
        stack<double> st;

        // Process cars from right to left (closest -> farthest)
        for (int i = n - 1; i >= 0; i--) {

            // Time needed for current car to reach target.
            double time =
                (double)(target - cars[i].first) / cars[i].second;

            // If current car arrives later than the fleet ahead,
            // it cannot catch that fleet and forms a new fleet.
            if (st.empty() || time > st.top()) {
                st.push(time);
            }

            // Otherwise:
            // time <= st.top()
            // Current car catches the fleet ahead before (or at)
            // the target, so it becomes part of that fleet.
            // No need to push anything.
        }

        // Each entry in the stack represents one fleet.
        return st.size();
    }
};
```
