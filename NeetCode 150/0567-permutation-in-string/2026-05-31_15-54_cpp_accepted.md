## Solution

```cpp
class Solution {
public:
    bool checkInclusion(string s1, string s2) {
        // Use constant size sliding window + Hashmap for frequency + matches variable for constant time lookup

        int m = s1.size();
        int n = s2.size();

        if(m > n)
            return false;

        vector<int> firstStringHash(26, 0);
        vector<int> secondStringHash(26, 0);
        int matches = 0;

        // Fill frequency of first string
        for(char ch: s1)
            firstStringHash[ch - 'a']++;

        // First window
        for(int i=0; i<m; i++){
            secondStringHash[s2[i] - 'a']++;
        }

        // Initial match count of s1 and s2
        for(int i=0; i<26; i++){    // 26 since only lowercase english characters
            if(firstStringHash[i] == secondStringHash[i])
                matches++;
        }

        if(matches == 26)
            return true;

        for(int i = m; i < n; i++) {
            int rightCharInd = s2[i] - 'a';
            int leftCharInd  = s2[i - m] - 'a';

            // Add right character
            if(firstStringHash[rightCharInd] == secondStringHash[rightCharInd])
                matches--;

            secondStringHash[rightCharInd]++;

            if(firstStringHash[rightCharInd] == secondStringHash[rightCharInd])
                matches++;

            // Remove left character
            if(firstStringHash[leftCharInd] == secondStringHash[leftCharInd])
                matches--;

            secondStringHash[leftCharInd]--;

            if(firstStringHash[leftCharInd] == secondStringHash[leftCharInd])
                matches++;

            if(matches == 26)
                return true;
        }

        return false;
    }
};
```
