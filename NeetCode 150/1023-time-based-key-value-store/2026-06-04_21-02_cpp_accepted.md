## Solution

```cpp
// Use unordered_map<key, vector<pair<timestamp, value>>>
// Use Binary search to find the value at given timestamp. BS works because of increasing nature/input of timestamps.
// Therefore, it is naturally sorted
class TimeMap {
    // {timestamp, valueString}
    typedef pair<int, string> P;
private:
    unordered_map<string, vector<P>> hashMap;

    string findValue(string& key, int timestamp){
        vector<P>& vec = hashMap[key];

        int low = 0, high = vec.size() - 1;
        int ans = -1;

        while(low <= high){
            int mid = low + (high - low) / 2;

            if(vec[mid].first == timestamp){
                ans = mid;
                break;
            }

            else if(vec[mid].first < timestamp){
                low = mid + 1;
                ans = mid;
            }else{
                high = mid - 1;
            }
        }

        return ans == -1 ? "" : vec[ans].second;
    }

public:
    TimeMap() {
        
    }
    
    void set(string key, string value, int timestamp) {
        hashMap[key].push_back({timestamp, value});
    }
    
    string get(string key, int timestamp) {
        if(hashMap.find(key) == hashMap.end())
            return "";

        // Use binary search to find the value based on the timestamp
        return findValue(key, timestamp);
    }
};

/**
 * Your TimeMap object will be instantiated and called as such:
 * TimeMap* obj = new TimeMap();
 * obj->set(key,value,timestamp);
 * string param_2 = obj->get(key,timestamp);
 */
```
