## Solution

```cpp
class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        // Use hashmap aproach. Keep iterating on i loop and do a 2-sum on j, k loops
        
        int n = nums.size();

        // Sorted set of vector
        set<vector<int>> triplets;

        for(int i=0; i<n; i++){
            
            // Better to skip already processed numbers eligible for triplets
            if(i>0 && nums[i] == nums[i-1]) continue;

            // Mimicing 2 sum
            unordered_set<int> hashSet;
            for(int j=i+1; j<n; j++){
                int target = -(nums[i] + nums[j]);
                if(hashSet.count(target) > 0){
                    vector<int> temp = {nums[i], nums[j], target};
                    sort(temp.begin(), temp.end()); // Sort and insert => to correctly remove the duplicates

                    triplets.insert(temp);
                }

                hashSet.insert(nums[j]);
            }
        }

        return vector<vector<int>>(triplets.begin(), triplets.end());
    }
};
```
