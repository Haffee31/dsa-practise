## Solution

```cpp
class Solution {
public:
    int characterReplacement(string s, int k) {
        vector<int> hash(26, 0);
        int longestSubstringLen = 0;
        int maxFreq = 0;
        int left = 0, right = 0;

        while(right < s.size()){
            // Put the current char into hash and update maxFreq
            hash[s[right] - 'A']++;
            maxFreq = max(maxFreq, hash[s[right] - 'A']);

            int windowSize = right - left + 1;
            int replacementsNeeded = windowSize - maxFreq;
            cout << "A" << endl;

            // Remove the left character, update hash freq but not max freq
            // maxFreq can go as stale. This will not update the answer wrongly, 
            // because we are decreasing the window size which will not allow the answer to grow incorrectly.
            if(windowSize - maxFreq > k){   // Using if instead of while here. Since it will be executed atmost once. Because decreasing window size will make it valid
                hash[s[left] - 'A']--;
                left++;

                windowSize = right - left + 1;
                // Do not update the maxFreq. It can go stale
            }

            if(windowSize - maxFreq <= k){
                longestSubstringLen = max(longestSubstringLen, windowSize);
            }

            right++;
        }

        return longestSubstringLen;
    }
};
```
