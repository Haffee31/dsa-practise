## Solution

```cpp
class Solution {
public:
    int largestRectangleArea(vector<int>& heights) {
        // Idea: At every bar, area[i] = heights[i] * (PSE[i] - NSE[i] - 1); 
        // Instead of calculating PSE, NSE, area of each bar separately, we calculate everything in a single iteration 
        // We will use the PSE iteration
        int n = heights.size();
        int maxArea = 0;

        // Monotonic increasing stack storing indices.
        // For every index in the stack:
        // heights[stack[0]] <= heights[stack[1]] <= ...
        stack<int> st;

        for (int i = 0; i < n; i++) {

            // Current bar becomes the Next Smaller Element (NSE)
            // for all taller bars in the stack.
            while (!st.empty() && heights[st.top()] > heights[i]) {
                int height = heights[st.top()];
                st.pop();

                int nse = i;    // IMP: NSE is current index
                int pse = st.empty() ? -1 : st.top();

                int width = nse - pse - 1;
                int area = height * width;

                maxArea = max(maxArea, area);
            }

            st.push(i);
        }

        // Process bars that do not have a smaller element on the right.
        while (!st.empty()) {
            int height = heights[st.top()];
            st.pop();

            int nse = n;    // IMP: NSE is n here
            int pse = st.empty() ? -1 : st.top();

            int width = nse - pse - 1;
            int area = height * width;

            maxArea = max(maxArea, area);
        }

        return maxArea;
    }
};
```
