## Solution

```cpp
class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        // Use prefix and suffix arrays to find the products on left and right
        // As optimal approach, use result array itself as prefix and suffix. Keep variables for running products

        int n = nums.size();

        vector<int> result(n, 1);
        
        int prefixProduct = 1;
        for(int i=1; i<n; i++){
            prefixProduct = prefixProduct * nums[i-1];
            result[i] = prefixProduct;
        }

        int suffixProduct = 1;
        for(int i=n-2; i>=0; i--){
            suffixProduct = suffixProduct * nums[i+1];
            result[i] = result[i] * suffixProduct;
        }

        return result;
    }
};
```
