## Solution

```cpp
class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        // Using Hashmap and bucket sort where i is freqCount

        int n = nums.size();
        unordered_map<int, int> mapNumToCount;
        for(int num: nums)
            mapNumToCount[num]++;

        vector<vector<int>> bucket(n+1);
        for(auto [num, count]: mapNumToCount){
            bucket[count].push_back(num);
        }

        vector<int> topKFreqElements;
        for(int i=n; i>=0; i--){
            for(int num: bucket[i]){
                topKFreqElements.push_back(num);
                k--;
                if(k == 0)
                    return topKFreqElements;
            }
        }

        return topKFreqElements;
    }
};
```
