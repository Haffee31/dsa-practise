## Solution

```cpp
class Solution {
public:
    vector<int> twoSum(vector<int>& numbers, int target) {
        // Use 2 pointer approach - i at start, j at end
        // if sum == target => found required pair
        // If sum > target => needs to make it less by moving j--
        // If sum < target => needs to increase, by moving i++

        int n = numbers.size();

        int i = 0;
        int j = n-1;

        while(i < j){
            int sum = numbers[i] + numbers[j];

            if(sum == target)
                return {i+1, j+1}; // Since 1-indexed array

            if(sum > target)
                j--;
            else
                i++;
        }

        return {-1, -1};
    }
};
```
