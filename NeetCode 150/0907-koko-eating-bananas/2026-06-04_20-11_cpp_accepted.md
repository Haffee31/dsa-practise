## Solution

```cpp
class Solution {
public:
    bool isPossible(int speed, vector<int>& piles, int h){
        int totalHours = 0;
        for(int pile: piles){
            int currentHours = pile/speed;
            
            if(pile%speed != 0) // Will take an extra hour to eat the remaining bananas from the pile
                currentHours++;

            totalHours += currentHours;
            if(totalHours > h)  // Early exit when total hours needed exceeeds the given hours
                return false;
        }

        return true;
    }

    int minEatingSpeed(vector<int>& piles, int h) {
        // Using binary search

        int maxi = INT_MIN;
        for(int pile: piles)
            maxi = max(maxi, pile);

        int low = 1, high = maxi;
        int ans = maxi;
        while(low <= high){
            int mid = low + (high - low) / 2;

            if(isPossible(mid, piles, h)){
                ans = min(ans, mid);
                high = mid - 1;
            }else{
                low = mid + 1;
            }
        }

        return ans;
    }
};
```
